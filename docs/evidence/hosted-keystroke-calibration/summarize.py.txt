import collections
import json
import math
import statistics
import sys

record = json.load(open(sys.argv[1]))
budgets = json.load(open('perf/budgets.json'))
rows = record['rows']
unique = [r for r in rows if 'carriedFrom' not in r]
carried = [r for r in rows if 'carriedFrom' in r]
surfaces = ['editor-empty', 'editor-page-5000-blocks', 'empty', 'page-5000-blocks', 'layout-delay']
metrics = [('palette p50', 'palette', 'p50'), ('palette p95', 'palette', 'p95'),
           ('scroll work p95', 'scroll', 'p95'), ('scroll work max', 'scroll', 'workMax'),
           ('scroll longest', 'scroll', 'longest'), ('scroll over33 count', 'scroll', 'over33')]
for r in unique:
    if r.get('palette', {}).get('p95') is not None:
        assert r['palette']['keys'] == 55
    for s in r.get('typing', {}).values():
        assert s['samples'] == 405
    if r.get('scroll', {}).get('longest') is not None:
        assert r['scroll']['frames'] > 200
assert len({(r['run'], r.get('started')) for r in unique if r['job']}) == sum(bool(r['job']) for r in unique)

def value(row, section, key):
    if section == 'typing':
        return row.get('typing', {}).get(key, {}).get('p95')
    return row.get(section, {}).get(key)

def result_load(row, section, key):
    if section == 'typing':
        return row['typing'][key]['load1']
    return row[section]['load1']

def span(values):
    return 'missing' if not values else f'{min(values):g} / {statistics.median(values):g} / {max(values):g}'

def correlation(pairs):
    if len(pairs) < 2:
        return 'n/a'
    xs, ys = zip(*pairs)
    mx, my = statistics.mean(xs), statistics.mean(ys)
    xx = sum((x - mx) ** 2 for x in xs)
    yy = sum((y - my) ** 2 for y in ys)
    if not xx or not yy:
        return 'n/a'
    r = sum((x-mx)*(y-my) for x, y in pairs) / math.sqrt(xx*yy)
    return f'{r:+.3f}, {100*r*r:.1f}%'

print('## Collection and metric ranges\n')
print(f"{record['runs']} CI runs produced {len(rows)} attempt rows. {len(carried)} rows carry forward an earlier job's start and measurements.")
print(f"There are {len(unique)} rows after counting those jobs once, including missing jobs. Log states are {dict(collections.Counter(r.get('log', 'no-job') for r in unique))}.\n")
errors = [(r['job'], r['summaryErrors']) for r in unique if r.get('summaryErrors')]
print(f'Explicit missing-summary errors are {errors}. The collector exits nonzero when this list is nonempty. Other parsed metrics remain available.\n')
print('| Metric, ms except count | Unique jobs | All min / median / max | Main min / median / max | PR min / median / max |')
print('|---|---:|---|---|---|')
for name, section, key in metrics:
    measured = [r for r in unique if value(r, section, key) is not None]
    print(f"| {name} | {len(measured)} | {span([value(r, section, key) for r in measured])} | {span([value(r, section, key) for r in measured if r['branch']=='main'])} | {span([value(r, section, key) for r in measured if r['branch']!='main'])} |")
for surface in surfaces:
    measured = [r for r in unique if r.get('typing', {}).get(surface)]
    p95 = lambda r: r['typing'][surface]['p95']
    print(f"| {surface} | {len(measured)} | {span([p95(r) for r in measured])} | {span([p95(r) for r in measured if r['branch']=='main'])} | {span([p95(r) for r in measured if r['branch']!='main'])} |")

print('\n## Gate sensitivity and replay\n')
print('All four palette and scroll ceilings use `ceil(1.5 * worst observed)`. The median runner is the median of all unique measured jobs, including failures. A strict exceedance fails, so the first whole-millisecond failing value is one millisecond above the ceiling.\n')
print('| Gate | Ceiling, ms | Median, ms | Fails above added ms | Shift to first whole-ms failure | First failing value / median | Replay failures / measured |')
print('|---|---:|---:|---:|---:|---:|---:|')
gates = [('palette p50', 'palette', 'p50', 'hostedAppOwnedP50Ms'),
         ('palette p95 backstop', 'palette', 'p95', 'hostedAppOwnedP95Ms'),
         ('scroll work p95', 'scroll', 'p95', 'hostedWorkP95Ms'),
         ('scroll interval backstop', 'scroll', 'longest', 'hostedMaxFrameMs')]
for name, section, key, budget in gates:
    values = [value(r, section, key) for r in unique if value(r, section, key) is not None]
    ceiling = budgets[section][budget]['value']
    assert ceiling == math.ceil(1.5 * max(values))
    median = statistics.median(values)
    shift = math.floor(ceiling) + 1 - median
    print(f'| {name} | {ceiling} | {median:g} | {ceiling-median:g} | +{shift:g} | {(median+shift)/median:.2f}x | {sum(v>ceiling for v in values)}/{len(values)} |')
complete = [r for r in unique if all(value(r, section, key) is not None for _, section, key, _ in gates)]
old = [r for r in complete if r['palette']['p95'] > 40 or r['scroll']['longest'] > 50]
new = [r for r in complete if any(value(r, section, key) > budgets[section][budget]['value'] for _, section, key, budget in gates)]
print(f'\nPalette and scroll replay false failures are {len(old)}/{len(complete)} with the former gates and {len(new)}/{len(complete)} with these four gates. This labels identical-code exceedances as false failures for calibration; it does not turn historical job conclusions into passes.')
for surface in surfaces[:2]:
    measured = [r for r in unique if r.get('typing', {}).get(surface)]
    failures = [r for r in measured if r['typing'][surface]['p95'] > budgets['keystroke']['hostedAppOwnedP95Ms']['value']]
    print(f"\nUnchanged typing gate, {surface}: {len(failures)}/{len(measured)} exceedances. Jobs: {', '.join(str(r['job']) for r in failures) or 'none'}.")

print('\n## Load associations and repeated checkouts\n')
print('Each correlation cell gives signed Pearson r and R squared. The four repeated palette and scroll checkouts are reruns after a failure, so selection favors noisy groups. These associations do not identify a causal limiter.\n')
print('| Metric | Wait-entry r, R squared | Result-load r, R squared | Repeated jobs / groups | Within-checkout variation / total | Centered wait-entry r, R squared |')
print('|---|---:|---:|---:|---:|---:|')
for name, section, key in metrics + [(s, 'typing', s) for s in surfaces]:
    measured = [r for r in unique if value(r, section, key) is not None and r.get('wait') and r.get('checkout')]
    groups = collections.defaultdict(list)
    for r in measured:
        groups[r['checkout']].append(r)
    repeated = [g for g in groups.values() if len(g) > 1]
    subset = [r for g in repeated for r in g]
    val = lambda r: value(r, section, key)
    within = sum(sum((val(r)-statistics.mean(val(s) for s in g))**2 for r in g) for g in repeated)
    total = sum((val(r)-statistics.mean(val(s) for s in subset))**2 for r in subset)
    centered = [(r['wait']['startLoad1']-statistics.mean(s['wait']['startLoad1'] for s in g), val(r)-statistics.mean(val(s) for s in g)) for g in repeated for r in g]
    share = 'n/a' if not total else f'{100*within/total:.1f}%'
    print(f"| {name} | {correlation([(r['wait']['startLoad1'],val(r)) for r in measured])} | {correlation([(result_load(r, section, key),val(r)) for r in measured])} | {len(subset)} / {len(repeated)} | {share} | {correlation(centered)} |")

print('\n| Checkout SHA | Head SHA | Wait-entry loads | Palette p50 / p95, ms | Scroll work p95 / max / longest, ms | Typing p95 by surface, ms |')
print('|---|---|---|---|---|---|')
groups = collections.defaultdict(list)
for row in unique:
    if row.get('checkout') and row.get('palette', {}).get('p95') is not None:
        groups[row['checkout']].append(row)
fmt = lambda v: 'missing' if v is None else str(v)
for checkout, group in groups.items():
    if len(group) < 2:
        continue
    sequence = lambda f: ', '.join(fmt(f(r)) for r in group)
    typing = '; '.join(s + ' ' + sequence(lambda r,s=s:r.get('typing',{}).get(s,{}).get('p95')) for s in surfaces)
    palette = '; '.join('/'.join(fmt(r['palette'][k]) for k in ['p50', 'p95']) for r in group)
    scroll = '; '.join('/'.join(fmt(r['scroll'][k]) for k in ['p95', 'workMax', 'longest']) for r in group)
    print(f"| `{checkout}` | `{group[0]['head']}` | {sequence(lambda r:r['wait']['startLoad1'])} | {palette} | {scroll} | {typing} |")
waits = [r['wait'] for r in unique if r.get('wait')]
print('\n## Quiet-machine wait\n')
print(f"{len(waits)} unique jobs logged a wait. Entry load min / median / max was {span([w['startLoad1'] for w in waits])}. Exit load was {span([w['endLoad1'] for w in waits])}.")
print(f"Load fell in {sum(w['endLoad1']<w['startLoad1'] for w in waits)} jobs. {sum(w['endLoad1']<2.5 for w in waits)} reached the quiet threshold. {sum(w['seconds']>=120 for w in waits)} exhausted at least 120 seconds.")

print('\n## Extracted attempt table\n')
print('All times are milliseconds except wait duration in seconds. `missing` means no compatible summary was logged, not zero. Wait is entry load, exit load, duration. Palette is p50, p95, keys, result load. Scroll is work p95, work max, longest rAF interval, over33 count, frames, result load. Typing columns are pooled app-owned p95. Carried rows retain their measurements and the collector marks their original attempt.\n')
print('| Run / attempt | Branch | Head SHA | Checkout SHA | Job result / log | Wait | Palette | Scroll | editor-empty | editor-page-5000-blocks | empty | page-5000-blocks | layout-delay |')
print('|---|---|---|---|---|---|---|---|---:|---:|---:|---:|---:|')
for r in rows:
    job = f"[{r['status']}](https://github.com/aelomi/aelomi/actions/runs/{r['run']}/job/{r['job']})" if r['job'] else r['status']
    state = 'carried from attempt ' + str(r['carriedFrom']) if 'carriedFrom' in r else r.get('log','no-job')
    if r.get('summaryErrors'):
        state += '; summary error ' + ', '.join(r['summaryErrors'])
    w = r.get('wait')
    wait = '/'.join(str(w[k]) for k in ['startLoad1','endLoad1','seconds']) if w else 'missing'
    pal = '/'.join(fmt(r.get('palette',{}).get(k)) for k in ['p50','p95','keys','load1'])
    scroll = '/'.join(fmt(r.get('scroll',{}).get(k)) for k in ['p95','workMax','longest','over33','frames','load1'])
    typing = ' | '.join(fmt(r.get('typing',{}).get(s,{}).get('p95')) for s in surfaces)
    print(f"| [{r['run']}/{r['attempt']}](https://github.com/aelomi/aelomi/actions/runs/{r['run']}/attempts/{r['attempt']}) | {r['branch']} | `{r['head']}` | {r.get('checkout','missing')} | {job}, {state} | {wait} | {pal} | {scroll} | {typing} |")
