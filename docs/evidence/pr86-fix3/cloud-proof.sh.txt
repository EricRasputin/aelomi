#!/usr/bin/env bash
set -euo pipefail
umask 077
S=.claude/skills/verify-aelomi/scripts
BASE=$(mktemp -d /private/tmp/ae86-r3-XXXX)
printf '%s\n' "$BASE" > .local-gate/fix3/cloud-base.txt
: > .local-gate/fix3/cloud-runs.tsv
mkdir -p "$BASE/local" "$BASE/cloud/Aelomi Notes" "$BASE/cloud/Aelomi Dev" "$BASE/Library"
printf 'synthetic pages stay intact\n' > "$BASE/cloud/Aelomi Notes/sentinel"
printf 'synthetic shared pages stay intact\n' > "$BASE/cloud/Aelomi Dev/sentinel"
ln -s "$BASE/cloud/Aelomi Notes" "$BASE/local/Aelomi"
ln -s "$BASE/cloud" "$BASE/Library/Application Support"
RUN=
OWN_LOCK=0
OLD_DARK=
release_lock() {
  if [[ "$OWN_LOCK" == 1 && -f /tmp/aelomi-ui.lock/owner && "$(awk '{print $1}' /tmp/aelomi-ui.lock/owner)" == "$$" ]]; then
    rm -rf /tmp/aelomi-ui.lock
  fi
  OWN_LOCK=0
}
finish() {
  if [[ -n "$OLD_DARK" && "$OWN_LOCK" == 1 ]]; then
    osascript -e "tell application \"System Events\" to tell appearance preferences to set dark mode to $OLD_DARK" >/dev/null || true
  fi
  if [[ -f .local-gate/fix3/clipboard-before.txt && "$OWN_LOCK" == 1 ]]; then
    pbcopy < .local-gate/fix3/clipboard-before.txt
    rm .local-gate/fix3/clipboard-before.txt
  fi
  release_lock
  if [[ -n "$RUN" ]]; then "$S/app.sh" cleanup "$RUN" >/dev/null || true; fi
}
trap finish EXIT INT TERM
take_lock() {
  local started
  started=$(date +%s)
  until mkdir /tmp/aelomi-ui.lock 2>/dev/null; do
    if [[ -f /tmp/aelomi-ui.lock/owner ]] && (( $(date +%s) - $(stat -f %m /tmp/aelomi-ui.lock/owner) > 1800 )); then
      rm -rf /tmp/aelomi-ui.lock
    fi
    if (( $(date +%s) - started > 2700 )); then echo 'INCONCLUSIVE: keyboard lock unavailable for 45 minutes' >&2; exit 1; fi
    sleep 15
  done
  printf '%s %s E-fix3-cloud\n' "$$" "$(date -u +%FT%TZ)" > /tmp/aelomi-ui.lock/owner
  OWN_LOCK=1
}
for variant in leaf shared override; do
  case "$variant" in
    leaf) CONFIGURED="$BASE/local/Aelomi/" ;;
    shared) CONFIGURED="$BASE/Library/Application Support/Aelomi Dev" ;;
    override) CONFIGURED="$BASE/cloud/Aelomi Dev" ;;
  esac
  RUN=$(AELOMI_DATA_DIR="$CONFIGURED" AELOMI_CLOUD_ROOTS="$BASE/cloud" AELOMI_DEV_PORT=55086 "$S/app.sh" launch dev)
  printf '%s\t%s\n' "$variant" "$RUN" >> .local-gate/fix3/cloud-runs.tsv
  take_lock
  OLD_DARK=$(osascript -e 'tell application "System Events" to tell appearance preferences to get dark mode')
  "$S/app.sh" doctor "$RUN"
  "$S/app.sh" tree "$RUN" > "$RUN/cloud-tree.txt"
  for appearance in dark light; do
    if [[ "$appearance" == dark ]]; then DARK=true; else DARK=false; fi
    osascript -e "tell application \"System Events\" to tell appearance preferences to set dark mode to $DARK" >/dev/null
    "$S/app.sh" screenshot "$RUN" "cloud-$variant-$appearance"
  done
  if [[ "$variant" == leaf ]]; then
    pbpaste > .local-gate/fix3/clipboard-before.txt
    "$S/app.sh" focus "$RUN" 'In Finder, choose Go > Go to Folder (⇧⌘G) and paste:'
    if [[ "${AELOMI_CHIP_POINTER:-0}" == 1 ]]; then
      touch "$RUN/ready-for-chip-pointer"
      POINTER_STARTED=$(date +%s)
      until [[ -f "$RUN/chip-pointer-done" ]]; do
        if (( $(date +%s) - POINTER_STARTED > 300 )); then echo "INCONCLUSIVE: pointer proof timed out" >&2; exit 1; fi
        sleep 1
      done
    fi
    "$S/app.sh" keys "$RUN" a command
    "$S/app.sh" keys "$RUN" c command
    test "$(pbpaste)" = "$BASE/local"
    printf 'keyboard path copy matched the configured parent\n' >> "$RUN/proof.txt"
    pbcopy < .local-gate/fix3/clipboard-before.txt
    rm .local-gate/fix3/clipboard-before.txt
  fi
  osascript -e "tell application \"System Events\" to tell appearance preferences to set dark mode to $OLD_DARK" >/dev/null
  OLD_DARK=
  release_lock
  "$S/app.sh" cleanup "$RUN" >/dev/null
  RUN=
done
python3 - "$BASE" <<'PY'
from pathlib import Path
import sys
base=Path(sys.argv[1])
for name, expected in [('Aelomi Notes', b'synthetic pages stay intact\n'), ('Aelomi Dev', b'synthetic shared pages stay intact\n')]:
 target=base/'cloud'/name
 assert sorted(p.name for p in target.iterdir())==['sentinel']
 assert (target/'sentinel').read_bytes()==expected
assert (base/'local/Aelomi').is_symlink()
assert (base/'Library/Application Support').is_symlink()
print('PASS: all three variants preserved exact sentinel bytes, entries, and links')
PY
