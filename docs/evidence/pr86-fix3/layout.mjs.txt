import { writeFile } from "node:fs/promises";

export default async function ({ page, expect, shot, out }) {
  await page.addInitScript(() => {
    const query = new URL(location.href).searchParams;
    const worst = query.get("data") === "worst";
    const variant = query.get("variant") ?? "leaf";
    const configuredParent = "/Users/test/Library/Application Support";
    const resolvedParent = worst
      ? "/Users/test/Library/CloudStorage/Dropbox/Projects/Customer-Feedback-From-Enterprise-Onboarding/Research-and-Release-Planning/研究とメモ🦊/International-Collaboration-Archive"
      : "/Users/test/Library/Mobile Documents";
    const configuredName = worst
      ? "Aelomi-International-Research-And-Release-Planning-Notes"
      : "Aelomi";
    const resolvedName = worst
      ? "Q3-Board-Deck-FINAL-Revised-2026-Approved-By-Legal-研究🦊"
      : "Aelomi Notes";
    const configured = `${configuredParent}/${configuredName}`;
    const resolved = `${resolvedParent}/${variant === "shared" ? configuredName : resolvedName}`;
    const recovery =
      variant === "leaf"
        ? {
            LinkAtDataFolder: {
              configured_parent: configuredParent,
              configured_name: configuredName,
              resolved_parent: resolvedParent,
              resolved_name: resolvedName,
            },
          }
        : variant === "shared"
          ? { SharedLink: { link: configuredParent } }
          : "Override";
    window.__aelomiMockSetup = {
      storageStatus: {
        CloudPath: {
          configured: variant === "override" ? resolved : configured,
          resolved,
          recovery,
        },
      },
    };
    window.addEventListener("DOMContentLoaded", () => {
      const control = document.createElement("select");
      for (const name of ["Demo data", "Worst case"]) {
        const option = document.createElement("option");
        option.textContent = name;
        option.value = name === "Worst case" ? "worst" : "demo";
        control.append(option);
      }
      control.value = worst ? "worst" : "demo";
      control.setAttribute("aria-label", "Proof data");
      control.style.cssText =
        "position:fixed;bottom:4px;right:4px;z-index:1000;font:12px system-ui";
      control.onchange = () => {
        query.set("data", control.value);
        location.search = query.toString();
      };
      document.body.append(control);
    });
  });
  const results = [];
  for (const variant of ["leaf", "shared", "override"]) {
    for (const data of ["demo", "worst"]) {
      for (const colorScheme of ["dark", "light"]) {
        await page.emulateMedia({ colorScheme });
        await page.goto(`/?variant=${variant}&data=${data}`);
        await page.addStyleTag({
          content: ".storage-refusal-path { field-sizing: fixed !important; }",
        });
        await expect(
          page.getByRole("heading", {
            name: "Move your data folder out of cloud storage",
          }),
        ).toBeVisible();
        for (const width of [320, 480, 960, 1440]) {
          await page.setViewportSize({ width, height: 640 });
          const layout = await page.evaluate(() => ({
            fieldSizing: CSS.supports("field-sizing", "content"),
            overflow:
              document.documentElement.scrollWidth >
              document.documentElement.clientWidth,
            clipped: Array.from(document.querySelectorAll("textarea"))
              .filter(
                (chip) =>
                  chip.scrollHeight > chip.clientHeight + 1 ||
                  chip.scrollWidth > chip.clientWidth + 1,
              )
              .map((chip) => ({
                value: chip.value,
                height: chip.clientHeight,
                scrollHeight: chip.scrollHeight,
              })),
          }));
          expect(layout).toEqual({
            fieldSizing: true,
            overflow: false,
            clipped: [],
          });
          results.push({ variant, data, colorScheme, width, ...layout });
          if (
            (width === 960 && colorScheme === "dark") ||
            (width === 320 && colorScheme === "light" && data === "worst")
          )
            await shot(`${variant}-${data}-${colorScheme}-${width}`);
        }
        await page.setViewportSize({ width: 480, height: 640 });
        await page.evaluate(() => {
          document.documentElement.style.fontSize = "200%";
        });
        expect(
          await page.evaluate(
            () =>
              document.documentElement.scrollWidth >
              document.documentElement.clientWidth,
          ),
        ).toBe(false);
        if (data === "worst" && colorScheme === "dark")
          await shot(`${variant}-worst-dark-200pct`);
      }
    }
  }
  await writeFile(`${out}/layout.json`, JSON.stringify(results, null, 2));
}
