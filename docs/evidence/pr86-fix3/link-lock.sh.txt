#!/usr/bin/env bash
set -euo pipefail
umask 077
STARTED=$(date +%s)
until mkdir /tmp/aelomi-ui.lock 2>/dev/null; do
  if [[ -f /tmp/aelomi-ui.lock/owner ]] && (( $(date +%s) - $(stat -f %m /tmp/aelomi-ui.lock/owner) > 1800 )); then rm -rf /tmp/aelomi-ui.lock; fi
  if (( $(date +%s) - STARTED > 2700 )); then echo 'INCONCLUSIVE: keyboard lock unavailable for 45 minutes' >&2; exit 1; fi
  sleep 15
done
printf '%s %s E-fix3-links\n' "$$" "$(date -u +%FT%TZ)" > /tmp/aelomi-ui.lock/owner
finish() {
  if [[ -f /tmp/aelomi-ui.lock/owner && "$(awk '{print $1}' /tmp/aelomi-ui.lock/owner)" == "$$" ]]; then
    if [[ -f .local-gate/fix3/clipboard-before-link.txt ]]; then pbcopy < .local-gate/fix3/clipboard-before-link.txt; rm .local-gate/fix3/clipboard-before-link.txt; fi
    osascript -e 'tell application "System Events" to key up shift' >/dev/null || true
    rm -rf /tmp/aelomi-ui.lock
  fi
}
trap finish EXIT INT TERM
pbpaste > .local-gate/fix3/clipboard-before-link.txt
printf 'LOCK_ACQUIRED %s\n' "$$"
ACQUIRED=$(date +%s)
while [[ ! -f .local-gate/fix3/release-link-lock ]]; do
  if (( $(date +%s) - ACQUIRED > 1200 )); then echo 'Lock time limit reached' >&2; exit 1; fi
  sleep 1
done
