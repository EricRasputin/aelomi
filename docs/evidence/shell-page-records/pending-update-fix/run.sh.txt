#!/usr/bin/env bash
set -euo pipefail
S=.claude/skills/verify-aelomi/scripts
wait_for_quit() {
    local proof_run=$1
    local proof_app_pid
    proof_app_pid=$(rg '^APP_PID=' "$proof_run/run.env")
    proof_app_pid=${proof_app_pid#APP_PID=}
    for ((attempt=0; attempt<100; attempt++)); do
        if ! kill -0 "$proof_app_pid" 2>/dev/null; then
            rg 'quit flush via=CompleteQuit pending=\[\]' "$proof_run/shell.log"
            return
        fi
        sleep 0.1
    done
    return 1
}
cargo nextest run -p aelomi-core -E 'test(a_record_edit_never_reencodes) | test(workspace::tests)'
RUN=$($S/app.sh launch dev)
$S/app.sh doctor "$RUN"
$S/app.sh screenshot "$RUN" before
$S/app.sh focus "$RUN" "Page body"
$S/app.sh keys "$RUN" " fixed proof"
$S/app.sh screenshot "$RUN" after
$S/app.sh keys "$RUN" q command
wait_for_quit "$RUN"
$S/app.sh cleanup "$RUN"
AGAIN=$(AELOMI_DATA_DIR="$RUN/data" $S/app.sh launch dev)
$S/app.sh doctor "$AGAIN"
$S/app.sh screenshot "$AGAIN" relaunch
$S/app.sh tree "$AGAIN"
$S/app.sh keys "$AGAIN" q command
wait_for_quit "$AGAIN"
$S/app.sh cleanup "$AGAIN"
pnpm check:all
