import datetime
import json
import os
import subprocess
import sys
import time
from pathlib import Path

run_id, head, out_name = sys.argv[1:4]
reruns = int(sys.argv[4]) if len(sys.argv) > 4 else 3
assert 1 <= reruns <= 3
out = Path(out_name).expanduser()
out.mkdir(parents=True, exist_ok=True)
lock = out / 'series-running.pid'
with lock.open('x') as handle:
    handle.write(str(os.getpid()))

def gh(*args):
    return subprocess.run(['gh', *args], text=True, capture_output=True, check=True).stdout

def snapshot():
    result = json.loads(gh('run', 'view', run_id, '--json', 'status,conclusion,headSha,attempt,jobs'))
    assert result['headSha'] == head, result['headSha']
    return result

def completed(min_attempt):
    deadline = time.monotonic() + 2700
    last = None
    while time.monotonic() < deadline:
        result = snapshot()
        state = (result['attempt'], result['status'])
        if state != last:
            print(datetime.datetime.now(datetime.timezone.utc).isoformat(), state, flush=True)
            last = state
        if result['attempt'] >= min_attempt and result['status'] == 'completed':
            return result
        time.sleep(15)
    raise TimeoutError('workflow did not finish in 45 minutes')

try:
    current = completed(1)
    for series in range(1, reruns + 1):
        job = next(job for job in current['jobs'] if job['name'] == 'cold-start')
        attempt = current['attempt'] + 1
        target = out / f'series-{series}'
        assert not target.exists(), target
        request = {'run': run_id, 'head': head, 'job': job['databaseId'], 'attempt': attempt}
        (out / f'request-{series}.json').write_text(json.dumps(request, indent=2) + '\n')
        print('request', series, request, flush=True)
        gh('run', 'rerun', run_id, '--job', str(job['databaseId']))
        current = completed(attempt)
        gh('run', 'download', run_id, '--name', 'cold-start-results', '--dir', str(target))
        job = next(job for job in current['jobs'] if job['name'] == 'cold-start')
        result = {'run': run_id, 'head': head, 'attempt': current['attempt'], 'job': job['databaseId'], 'conclusion': job['conclusion']}
        assert len(list(target.rglob('results.json'))) == 2
        for path in sorted(target.rglob('results.json')):
            data = json.loads(path.read_text())
            result[data['expectedVariant']] = {key: data[key] for key in ['endToEndMedian', 'navToPaintMedian', 'hostedPass']}
        (out / f'series-{series}.json').write_text(json.dumps(result, indent=2) + '\n')
        print('completed', series, json.dumps(result), flush=True)
finally:
    lock.unlink()
