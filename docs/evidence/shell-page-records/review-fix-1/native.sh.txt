wait_quit() {
    python3 - "$1/shell.log" <<'PY'
from pathlib import Path
import sys
import time
log = Path(sys.argv[1])
for _ in range(100):
    lines = log.read_text()
    if "quit flush via=Exit pending=[]" in lines:
        print("\n".join(line for line in lines.splitlines() if line.startswith(("quit ", "tree warm", "page rename"))))
        break
    time.sleep(.1)
else:
    raise SystemExit("quit did not finish")
PY
}
