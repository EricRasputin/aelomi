set -euo pipefail
S=.claude/skills/verify-aelomi/scripts
LOCK_OWNER="$$ pr84-fix2-self-proof"
RUN=
DATA=
release() {
  python3 - "$LOCK_OWNER" <<'PY'
from pathlib import Path
import sys
owner=Path('/tmp/aelomi-ui.lock/owner')
if owner.exists() and sys.argv[1] in owner.read_text():
    owner.unlink()
    owner.parent.rmdir()
PY
}
cleanup() {
  release
  [ -z "$DATA" ] || chmod u+w "$DATA/docs"
  [ -z "$RUN" ] || "$S/app.sh" cleanup "$RUN"
}
trap cleanup EXIT
acquire() {
  started=$(date +%s)
  until mkdir /tmp/aelomi-ui.lock 2>/dev/null; do
    [ $(( $(date +%s) - started )) -lt 2700 ] || { echo INCONCLUSIVE; exit 1; }
    python3 - <<'PY'
from pathlib import Path
import time
owner=Path('/tmp/aelomi-ui.lock/owner')
try:
    if time.time()-owner.stat().st_mtime > 1800:
        owner.unlink()
        owner.parent.rmdir()
except FileNotFoundError:
    pass
PY
    sleep 15
  done
  echo "$LOCK_OWNER $(date -u +%FT%TZ)" > /tmp/aelomi-ui.lock/owner
}
wait_exit() {
  (. "$RUN/run.env"; for _ in $(seq 75); do kill -0 "$APP_PID" 2>/dev/null || exit 0; sleep 0.2; done; exit 1)
}
finish() {
  wait_exit
  release
  rg 'tree warm pages=|quit |page rename|retry rename' "$RUN/shell.log"
  "$S/app.sh" cleanup "$RUN"
}

  echo "$RUN"
  acquire
  "$S/app.sh" doctor "$RUN"
}
