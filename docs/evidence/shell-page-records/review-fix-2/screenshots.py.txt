import hashlib
import json
import re
import subprocess
import sys
import urllib.request
from pathlib import Path

comment_id, output, *files = sys.argv[1:]
body = json.loads(subprocess.check_output(['gh', 'api', f'repos/aelomi/aelomi/issues/comments/{comment_id}']))['body']
urls = re.findall(r'!\[[^\]]*\]\((https://[^)]+)\)', body)
assert len(urls) == len(files), (urls, files)
out = Path(output)
out.mkdir(parents=True, exist_ok=True)
for source, url in zip(files, urls):
    source = Path(source)
    original = source.read_bytes()
    with urllib.request.urlopen(url, timeout=60) as response:
        downloaded = response.read()
    assert downloaded.startswith(b'\x89PNG\r\n\x1a\n'), url
    assert downloaded == original, source
    destination = out / source.name
    destination.write_bytes(downloaded)
    print(source.name, len(original), hashlib.sha256(original).hexdigest(), url)
