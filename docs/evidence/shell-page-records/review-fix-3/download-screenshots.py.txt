import hashlib
import json
import re
import subprocess
from pathlib import Path

root = Path('.audit/fix3')
comment = json.loads(subprocess.check_output(['gh','api','repos/aelomi/aelomi/issues/comments/6203886880'],text=True))
(root/'posted-comment.md').write_text(comment['body'])
record = Path('docs/evidence/shell-page-records/review-fix-3/README.md').read_text()
shots = {name:{'bytes':int(size),'sha256':sha} for name,size,sha in re.findall(r'\| `([^`]+\.png)` \| (\d+) \| `([0-9a-f]{64})` \|',record)}
images = re.findall(r'!\[([^\]]+)\]\((https://[^)]+)\)',comment['body'])
assert len(images) == len(shots) == 9, images
out = root/'downloaded';out.mkdir(exist_ok=True)
for name,url in images:
    assert name in shots, name
    path = out/name
    subprocess.run(['curl','--disable','--fail','--location','--silent','--show-error',url,'--output',str(path)],check=True)
    data = path.read_bytes()
    expected = shots[name]
    assert data.startswith(b'\x89PNG\r\n\x1a\n'), name
    assert len(data) == expected['bytes'], name
    assert hashlib.sha256(data).hexdigest() == expected['sha256'], name
    subprocess.run(['sips','-g','pixelWidth','-g','pixelHeight',str(path)],check=True,capture_output=True)
    print(name,len(data),expected['sha256'],'anonymous PNG match')
print(comment['html_url'])
