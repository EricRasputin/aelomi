#!/usr/bin/env bash
set -euo pipefail
S=.claude/skills/verify-aelomi/scripts
RUN=$($S/app.sh launch dev)
$S/app.sh doctor "$RUN"
$S/app.sh tree "$RUN"
$S/app.sh screenshot "$RUN" before
$S/app.sh focus "$RUN" "Page body"
$S/app.sh keys "$RUN" " seed proof"
$S/app.sh screenshot "$RUN" after
$S/app.sh keys "$RUN" q command
# Wait for the owned app PID from run.env to exit, then require pending=[] in shell.log.
$S/app.sh cleanup "$RUN"
AGAIN=$(AELOMI_DATA_DIR="$RUN/data" $S/app.sh launch dev)
$S/app.sh doctor "$AGAIN"
$S/app.sh screenshot "$AGAIN" relaunch
$S/app.sh tree "$AGAIN"
$S/app.sh keys "$AGAIN" q command
$S/app.sh cleanup "$AGAIN"
cargo nextest run --workspace
pnpm test
pnpm typecheck
pnpm check:all
