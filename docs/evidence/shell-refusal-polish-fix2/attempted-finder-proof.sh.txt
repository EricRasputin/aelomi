#!/usr/bin/env bash
set -euo pipefail
umask 077
OUT=.local-gate/e2-fix2
S=.claude/skills/verify-aelomi/scripts
BASE=$(cat "$OUT/base.txt")
CONFIGURED="$BASE/Recovery Destination/Aelomi"
RUN=
OWN_LOCK=0
OLD_DARK=
FIRST_TITLE=
SECOND_TITLE=
UNNAMED_NEW=0
release_lock() {
  if [[ "$OWN_LOCK" == 1 && -f /tmp/aelomi-ui.lock/owner && "$(awk '{print $1}' /tmp/aelomi-ui.lock/owner)" == "$$" ]]; then rm -rf /tmp/aelomi-ui.lock; fi
  OWN_LOCK=0
}
take_lock() {
  local started
  started=$(date +%s)
  until mkdir /tmp/aelomi-ui.lock 2>/dev/null; do
    if [[ -f /tmp/aelomi-ui.lock/owner ]] && (( $(date +%s) - $(stat -f %m /tmp/aelomi-ui.lock/owner) > 1800 )); then rm -rf /tmp/aelomi-ui.lock; fi
    if (( $(date +%s) - started > 2700 )); then echo 'INCONCLUSIVE: keyboard lock unavailable for 45 minutes'; exit 1; fi
    sleep 15
  done
  printf '%s %s PR90-fix2\n' "$$" "$(date -u +%FT%TZ)" > /tmp/aelomi-ui.lock/owner
  OWN_LOCK=1
}
close_title() {
  osascript - "$1" <<'AS'
on run argv
 tell application "System Events" to tell process "Finder"
  if exists window (item 1 of argv) then
   perform action "AXPress" of (first button of window (item 1 of argv) whose subrole is "AXCloseButton")
  end if
 end tell
end run
AS
}
close_finder() {
  if [[ "$UNNAMED_NEW" == 1 ]]; then
    osascript -e 'tell application "System Events" to tell process "Finder" to perform action "AXPress" of (first button of window 1 whose subrole is "AXCloseButton")'
    UNNAMED_NEW=0
  fi
  if [[ -n "$FIRST_TITLE" ]]; then close_title "$FIRST_TITLE"; FIRST_TITLE=; fi
  if [[ -n "$SECOND_TITLE" ]]; then close_title "$SECOND_TITLE"; SECOND_TITLE=; fi
}
finish() {
  if [[ "$OWN_LOCK" == 1 ]]; then
    close_finder || true
    if [[ -n "$OLD_DARK" ]]; then osascript -e "tell application \"System Events\" to tell appearance preferences to set dark mode to $OLD_DARK" >/dev/null || true; fi
    if [[ -f "$OUT/clipboard-before.txt" ]]; then pbcopy < "$OUT/clipboard-before.txt"; fi
  fi
  release_lock
  if [[ -n "$RUN" ]]; then "$S/app.sh" cleanup "$RUN" >/dev/null || true; fi
  ls -la "$HOME/Library/Application Support/Aelomi" > "$OUT/real-data-after.txt"
  diff -u "$OUT/real-data-before.txt" "$OUT/real-data-after.txt"
}
trap finish EXIT
trap 'exit 130' INT
trap 'exit 143' TERM
launch() {
  RUN=$(AELOMI_DATA_DIR="$1" AELOMI_CLOUD_ROOTS="$BASE/Proof Cloud" "$S/app.sh" launch dev)
  printf '%s\n' "$RUN" > "$OUT/$2-run.txt"
  take_lock
  "$S/app.sh" doctor "$RUN"
  if [[ -z "$OLD_DARK" ]]; then OLD_DARK=$(osascript -e 'tell application "System Events" to tell appearance preferences to get dark mode'); fi
}
quit_app() {
  local pid
  pid=$(sed -n 's/^APP_PID=//p' "$RUN/run.env")
  "$S/app.sh" keys "$RUN" q command
  for attempt in {1..40}; do if ! kill -0 "$pid" 2>/dev/null; then break; fi; sleep 0.1; done
  if kill -0 "$pid" 2>/dev/null; then "$S/app.sh" tree "$RUN"; echo 'FAIL: app did not quit'; exit 1; fi
  if grep -q '^storage status=Ready' "$RUN/shell.log"; then
    grep '^quit frontend ready' "$RUN/shell.log"
    grep '^quit draining via=ExitRequested' "$RUN/shell.log"
    grep '^quit confirmed via=drain' "$RUN/shell.log"
  else
    if grep -Eq '^quit frontend ready$|^quit draining|via=CompleteQuit' "$RUN/shell.log"; then echo 'FAIL: refused app registered a frontend drain'; exit 1; fi
    grep '^quit frontend ready skipped: storage refused' "$RUN/shell.log"
    grep '^quit flush via=ExitRequested pending=None' "$RUN/shell.log"
    echo 'PASS: refused app quit without a frontend drain or dialog'
  fi
}
cleanup_run() {
  osascript -e "tell application \"System Events\" to tell appearance preferences to set dark mode to $OLD_DARK" >/dev/null
  pbcopy < "$OUT/clipboard-before.txt"
  release_lock
  "$S/app.sh" cleanup "$RUN" >/dev/null
  RUN=
}
capture_variant() {
  local variant=$1
  osascript - "$(sed -n 's/^APP_PID=//p' "$RUN/run.env")" <<'AS'
on run argv
 tell application "System Events" to tell (first process whose unix id is (item 1 of argv as integer))
  set size of window 1 to {960, 1200}
 end tell
end run
AS
  for scheme in dark light; do
    if [[ "$scheme" == dark ]]; then dark=true; else dark=false; fi
    osascript -e "tell application \"System Events\" to tell appearance preferences to set dark mode to $dark"
    "$S/app.sh" screenshot "$RUN" "$variant-$scheme"
  done
}
copy_chip() {
  python3 - "$RUN/refusal-tree.txt" "$1" "$2" <<'PYCHIP' | pbcopy
import json,re,sys
from pathlib import Path
values=[]
for line in Path(sys.argv[1]).read_text().splitlines():
    match=re.fullmatch(r'\s*AXTextArea ("(?:[^"\\]|\\.)*") = ("(?:[^"\\]|\\.)*")',line)
    if match and json.loads(match[1])==sys.argv[2]: values.append(json.loads(match[2]))
assert values==[sys.argv[3]], values
print(values[0],end='')
PYCHIP
  test "$(pbpaste)" = "$2"
}
go_to_folder() {
  osascript - "$1" <<'AS'
on run argv
  set expectedName to item 1 of argv
  tell application "System Events" to tell process "Finder"
    set frontmost to true
    keystroke "g" using {shift down, command down}
    repeat 50 times
      if exists sheet 1 of window 1 then exit repeat
      delay 0.1
    end repeat
    keystroke "v" using {command down}
    key code 36
  end tell
  repeat 50 times
    tell application "System Events" to tell process "Finder"
      if (count windows) > 0 and (name of window 1) is expectedName then return
    end tell
    delay 0.1
  end repeat
  error "Finder did not reach the copied chip path"
end run
AS
}
