export default async function ({ page, expect, shot, out }) {
  await page.addInitScript(() => {
    const variant = new URL(location.href).searchParams.get('fixture') ?? 'demo';
    const name = variant === 'worst' ? 'InternationalCollaborationArchiveCustomerResearchAndProductPlanning2026研究とメモ🦊' : 'Aelomi';
    const parent = variant === 'worst' ? '/Users/test/Library/CloudStorage/Dropbox/International-Collaboration-Archive/Research/Planning/QuarterlyReports/2026/研究とメモ🦊' : '/Users/test/Library/CloudStorage/Support';
    const configured = '/Users/test/Library/Application Support';
    window.__aelomiMockSetup = { storageStatus: { CloudPath: {
      configured: `${configured}/${name}`, resolved: `${parent}/${name}`,
      recovery: variant === 'shared' ? { SharedLink: { link: parent } } : variant === 'override' ? 'Override' : { LinkAtDataFolder: {
        configured_parent: configured, configured_name: name, resolved_parent: parent, resolved_name: variant === 'worst' ? name + 'OriginalCloudFolder' : name,
      } },
    } } };
    addEventListener('DOMContentLoaded', () => {
      const toggle = document.createElement('nav');
      toggle.setAttribute('aria-label', 'Proof fixtures');
      toggle.style.cssText = 'position:fixed;bottom:0;left:50%;transform:translateX(-50%);z-index:100;background:#ddd;color:#111;font:12px system-ui;padding:4px;display:flex;gap:4px';
      for (const key of ['demo', 'worst', 'shared', 'override']) {
        const link = document.createElement('a');
        link.href = `/?fixture=${key}`;
        link.textContent = key === 'demo' ? 'Demo data' : key === 'worst' ? 'Worst case' : key;
        link.style.cssText = 'color:inherit;padding:3px;background:white;border-radius:4px';
        toggle.append(link);
      }
      document.body.append(toggle);
    });
  });
  for (const fixture of ['demo', 'worst', 'shared', 'override']) {
    for (const colorScheme of ['dark', 'light']) {
      await page.emulateMedia({ colorScheme });
      for (const width of [320, 960, 1440]) {
        await page.setViewportSize({ width, height: 900 });
        await page.goto(`/?fixture=${fixture}`);
        await expect(page.getByRole('heading', { name: 'Move your data folder out of cloud storage' })).toBeVisible();
        for (const fontSize of ['100%', '200%']) {
          await page.evaluate(size => { document.documentElement.style.fontSize = size; }, fontSize);
          expect(await page.evaluate(() => ({
            overflow: document.documentElement.scrollWidth > document.documentElement.clientWidth,
            clipped: Array.from(document.querySelectorAll('.storage-refusal-path')).filter(field => field.scrollWidth > field.clientWidth + 1 || field.scrollHeight > field.clientHeight + 1).map(field => field.value),
          }))).toEqual({ overflow: false, clipped: [] });
          await expect(page.getByText('Keep Aelomi open until the last step.', { exact: true })).toBeVisible();
          await page.getByText("Aelomi hasn't changed anything in your data folder.", { exact: true }).scrollIntoViewIfNeeded();
          await expect(page.getByText('Quit Aelomi and open it again.', { exact: true })).toBeVisible();
          if (width === 320 && fontSize === '200%' && true) await shot(`${fixture}-${colorScheme}-320-200`);
        }
      }
    }
  }
}
