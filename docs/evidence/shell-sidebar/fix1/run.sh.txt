#!/usr/bin/env bash
set -euo pipefail
# Run from the repository root with no other dev instance from this worktree.
E=docs/evidence/shell-sidebar/fix1
S=.claude/skills/verify-aelomi/scripts
mkdir -p .local-gate
cp docs/evidence/shell-sidebar/keyboard-lock.py.txt /tmp/aelomi-d1-lock.py
cp "$E/native-probe.mjs.txt" .local-gate/native-probe.ts
cp index.html .local-gate/index-before-native.html
D1_RUN=
D1_APPEARANCE=
finish() {
  cp .local-gate/index-before-native.html index.html
  rm -f .local-gate/native-probe.ts
  if [ -n "$D1_APPEARANCE" ]; then
    osascript -e "tell application \"System Events\" to tell appearance preferences to set dark mode to $D1_APPEARANCE" || true
  fi
  python3 /tmp/aelomi-d1-lock.py release
  if [ -n "$D1_RUN" ]; then bash "$S/app.sh" cleanup "$D1_RUN"; fi
}
trap finish EXIT
python3 - <<'PATCH'
from pathlib import Path
p=Path('index.html')
p.write_text(p.read_text().replace('</body>', '<script type="module" src="/.local-gate/native-probe.ts"></script>\n</body>'))
PATCH
D1_RUN=$(env -u AELOMI_DATA_DIR bash "$S/app.sh" launch dev --timeout 900)
bash "$S/app.sh" doctor "$D1_RUN"
python3 /tmp/aelomi-d1-lock.py
D1_APPEARANCE=$(osascript -e 'tell application "System Events" to tell appearance preferences to get dark mode')
bash "$S/app.sh" press "$D1_RUN" 'Fail record renames'
bash "$S/app.sh" focus "$D1_RUN" 'Page title'
bash "$S/app.sh" keys "$D1_RUN" a command
bash "$S/app.sh" keys "$D1_RUN" 'Pending native title'
bash "$S/app.sh" press "$D1_RUN" 'Read source title'
bash "$S/app.sh" tree "$D1_RUN" > "$D1_RUN/pending-title.ax.txt"
python3 - "$D1_RUN/pending-title.ax.txt" <<'ASSERT'
from pathlib import Path
import sys
s=Path(sys.argv[1]).read_text()
assert 'AXTextArea "Page title" = "Pending native title"' in s
assert 'SOURCE_TITLE=Getting started' in s
assert "This title couldn't" not in s
ASSERT
for appearance in true false; do
  osascript -e "tell application \"System Events\" to tell appearance preferences to set dark mode to $appearance"
  bash "$S/app.sh" screenshot "$D1_RUN" "pending-title-$appearance"
done
bash "$S/app.sh" press "$D1_RUN" 'Recover record renames'
bash "$S/app.sh" keys "$D1_RUN" q command
for attempt in $(seq 1 20); do
  if rg -q 'quit flush via=Exit pending=None' "$D1_RUN/shell.log"; then break; fi
  sleep 0.5
done
rg 'quit flush via=Exit pending=None' "$D1_RUN/shell.log"
python3 /tmp/aelomi-d1-lock.py release
bash "$E/read-records.sh.txt" "$D1_RUN/data" pending
printf 'Evidence: %s\n' "$D1_RUN"
