function run(argv) {
 const se = Application('System Events');
 const proc = se.processes.whose({ unixId: Number(argv[0]) })[0];
 proc.frontmost = true;
 se.keystroke('Native body text wraps through the scrollport while its title scrolls away. '.repeat(95));
 delay(20);
 for (let i = 0; i < 44; i++) { se.keyCode(126); delay(0.08); }
 delay(2);
 for (let i = 0; i < 12; i++) { se.keyCode(126); delay(0.08); }
 delay(2);
 se.keystroke(' caret near the top ');
 delay(2);
}
