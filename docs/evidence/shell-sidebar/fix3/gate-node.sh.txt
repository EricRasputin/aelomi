#!/bin/bash
set -euo pipefail
D1_GATE_REAL_NODE=${AELOMI_GATE_NODE:-}
if [ -z "$D1_GATE_REAL_NODE" ]; then
 D1_GATE_WRAPPER=$(cd "$(dirname "$0")" && pwd)/$(basename "$0")
 while IFS= read -r D1_GATE_CANDIDATE; do
  if [ "$D1_GATE_CANDIDATE" != "$D1_GATE_WRAPPER" ]; then D1_GATE_REAL_NODE=$D1_GATE_CANDIDATE; break; fi
 done < <(type -a -p node)
fi
if [ -z "$D1_GATE_REAL_NODE" ]; then echo 'Node executable is unavailable'; exit 1; fi
if [ "${1:-}" != perf/cold-start.ts ]; then exec "$D1_GATE_REAL_NODE" "$@"; fi
started=$(date +%s)
until mkdir /tmp/aelomi-ui.lock 2>/dev/null; do
 if [ -f /tmp/aelomi-ui.lock/owner ] && [ $(( $(date +%s) - $(stat -f %m /tmp/aelomi-ui.lock/owner) )) -gt 1800 ]; then rm -rf /tmp/aelomi-ui.lock; fi
 if [ $(( $(date +%s)-started )) -ge 2700 ]; then echo 'INCONCLUSIVE keyboard lock'; exit 1; fi
 sleep 15
done
token="$$ $(date -u +%FT%TZ) D1-fix3-cold-start"
printf '%s\n' "$token" > /tmp/aelomi-ui.lock/owner
finish() { if [ "$(cat /tmp/aelomi-ui.lock/owner 2>/dev/null)" = "$token" ]; then rm -rf /tmp/aelomi-ui.lock; fi; }
trap finish EXIT
"$D1_GATE_REAL_NODE" "$@"
