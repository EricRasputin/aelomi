export default async function ({ page, expect, shot }) {
 await page.addInitScript(() => { window.__aelomiMockSetup = { pages: [{id:'a',title:'Alpha'}] }; });
 await page.goto('/page/a');
 await expect(page.getByRole('textbox',{name:'Page title',exact:true})).toHaveValue('Alpha');
 await page.evaluate(() => window.__aelomiMock.storage.refuseRecords({Io:'held'}));
 await page.keyboard.press('Meta+n');
 const title=page.getByRole('textbox',{name:'Page title',exact:true});
 const full='Quarterly planning https://example.com/platform-infrastructure/research/王秀英-👩🏽‍💻-نور';
 await title.fill(full);
 await page.evaluate(() => window.__aelomiMock.storage.refuseRecords({Internal:'terminal create'}));
 await expect(title).toHaveJSProperty('readOnly',true);
 const row=page.locator('.sidebar-row[data-state="failed"]');
 const suffix=row.locator('.sidebar-failure');
 for(const scheme of ['light','dark']) {
  await page.emulateMedia({colorScheme:scheme});
  for(const width of [320,960,2560]) {
   await page.setViewportSize({width,height:720});
   await expect(suffix).toBeVisible();
   const box=await suffix.boundingBox();
   const parent=await row.boundingBox();
   expect(box.x+box.width).toBeLessThanOrEqual(parent.x+parent.width);
   expect(box.y+box.height).toBeLessThanOrEqual(720);
   await shot(`failed-row-${scheme}-${width}`);
  }
  await page.setViewportSize({width:960,height:720});
  await page.evaluate(() => { document.body.style.zoom='2'; });
  await shot(`failed-row-${scheme}-zoom200`);
  await page.evaluate(() => { document.body.style.zoom=''; });
 }
 await page.keyboard.press('Meta+k');
 await page.getByRole('combobox').fill('Quarterly');
 await expect(page.getByRole('option',{name:full,exact:true}).getByText('Not created',{exact:true})).toBeVisible();
 await shot('failed-row-palette-dark');
}
